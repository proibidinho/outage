---
# =============================================================================
# PLAYBOOK ÚNICO - VALIDAÇÃO / DEPLOY INFOBLOX DNS LINUX
# =============================================================================
# Variáveis do Survey:
#   - modo_validacao: validation | pos-validacao | deploy   (default: validation)
#       * validation    -> PRE-CHECK (comportamento atual, inalterado)
#       * pos-validacao -> POS-VALIDACAO da GMUD (somente leitura)
#       * deploy        -> ALTERAÇÃO EFETIVA do DNS. Executa os roles existentes
#                          (rhel_<dc> / sunos_<dc>) SOMENTE nos hosts classificados
#                          como "OK PARA ALTERAÇÃO" (mesmos critérios do pre-check)
#                          e revalida o resultado em seguida (somente leitura).
#                          Rollback: roles rhel_rollback / sunos_rollback.
#   - servers: Lista de IPs (um por linha)
#   - datacenter: Mackenzie | Lapa | Campinas
#   - email_destino: Email para enviar relatório
#   - rdm: Número da RDM (opcional)
# =============================================================================

- name: Adicionar servidores ao inventory
  hosts: localhost
  gather_facts: false

  vars:
    modo: "{{ modo_validacao | default('validation') }}"

  tasks:
    # Trava de segurança: um valor errado no Survey nunca deve cair em um modo
    # "silencioso" (principalmente agora que existe o modo deploy).
    - name: Validar parâmetros do Survey
      ansible.builtin.assert:
        that:
          - modo in ['validation', 'pos-validacao', 'deploy']
          - datacenter is defined
          - datacenter in ['Mackenzie', 'Lapa', 'Campinas']
          - servers is defined
          - servers.splitlines() | map('trim') | select() | list | length > 0
        fail_msg: >-
          Parametros invalidos no Survey.
          modo_validacao deve ser validation, pos-validacao ou deploy (recebido: {{ modo_validacao | default('nao definido') }});
          datacenter deve ser Mackenzie, Lapa ou Campinas (recebido: {{ datacenter | default('nao definido') }});
          servers nao pode estar vazio.
        quiet: true

    - name: Adicionar servidores ao grupo unix
      ansible.builtin.add_host:
        name: "{{ item | trim }}"
        groups: unix
      loop: "{{ servers.splitlines() | map('trim') | select() | unique | list }}"
      run_once: true

# =============================================================================
# PLAY 2: VALIDAR CONEXÃO SSH E PORTA 53
# =============================================================================
- name: Validar conectividade SSH e DNS
  hosts: unix
  gather_facts: false
  ignore_unreachable: true
  ignore_errors: true

  vars:
    # Modo de execução controlado pelo Survey do AAP
    modo: "{{ modo_validacao | default('validation') }}"
    # IPs DNS por datacenter (fonte de verdade = roles do infoblox.yaml)
    dns_ips:
      Mackenzie:
        - "10.54.44.13"
        - "10.53.22.101"
      Lapa:
        - "10.29.177.81"
        - "10.53.22.102"
      Campinas:
        - "10.54.44.12"
        - "10.29.177.82"
    # Sufixo dos roles por datacenter (rhel_<sufixo> / sunos_<sufixo>)
    dc_role_sufixo:
      Mackenzie: mck
      Lapa: lapa
      Campinas: cpa

  tasks:
    # -------------------------------------------------------------------------
    # ETAPA 1: TESTAR CONEXÃO SSH (compartilhado entre os dois modos)
    # -------------------------------------------------------------------------
    - name: SSH | Testar conexão SSH
      ansible.builtin.raw: echo "SSH_OK"
      register: ssh_test
      changed_when: false

    - name: SSH | Registrar status da conexão
      ansible.builtin.set_fact:
        ssh_ok: "{{ ssh_test.stdout is defined and 'SSH_OK' in ssh_test.stdout }}"
        ssh_erro_msg: "{{ ssh_test.msg | default('') }}"

    # -------------------------------------------------------------------------
    # ETAPA 2: DETECTAR SO (mesma lógica do infoblox.yaml)
    # -------------------------------------------------------------------------
    - name: SO | Detectar sistema operacional
      ansible.builtin.raw: cat /etc/redhat-release 2>/dev/null || cat /etc/system-release 2>/dev/null || uname -s 2>/dev/null
      register: so_detect
      when: ssh_ok | default(false)
      changed_when: false

    - name: SO | Registrar SO detectado (simplificado)
      ansible.builtin.set_fact:
        so_detectado: >-
          {%- if 'Red Hat' in so_detect.stdout -%}
            RHEL
          {%- elif 'SunOS' in so_detect.stdout -%}
            SunOS
          {%- elif 'CentOS' in so_detect.stdout -%}
            CentOS
          {%- elif 'Oracle' in so_detect.stdout -%}
            Oracle Linux
          {%- elif 'Ubuntu' in so_detect.stdout -%}
            Ubuntu
          {%- else -%}
            Linux
          {%- endif -%}
      when: ssh_ok | default(false)

    # -------------------------------------------------------------------------
    # ETAPA 3: TESTAR PORTA 53 (mesma lógica dos roles - nc/telnet/dig/nmap)
    # Compartilhado entre pre-check e pos-validacao.
    # -------------------------------------------------------------------------
    - name: DNS | Testar conectividade DNS - IP 1
      ansible.builtin.raw: |
        IP="{{ dns_ips[datacenter][0] }}"
        NC_OK=0
        TELNET_OK=0
        DIG_OK=0
        NMAP_OK=0

        if command -v nc >/dev/null 2>&1 && nc -h 2>&1 | grep -q '\-z'; then
          nc -z -w2 "$IP" 53 >/dev/null 2>&1 && NC_OK=1
        elif command -v telnet >/dev/null 2>&1; then
          telnet_output=$(echo quit | telnet "$IP" 53 2>/dev/null)
          echo "$telnet_output" | grep -q "Escape character" && TELNET_OK=1
        fi

        if command -v dig >/dev/null 2>&1; then
          dig_output=$(dig @"$IP" google.com +time=2 +tries=1 2>/dev/null)
          echo "$dig_output" | grep -q "ANSWER SECTION" && DIG_OK=1
        fi

        if command -v nmap >/dev/null 2>&1; then
          nmap_output=$(nmap -p 53 -Pn --host-timeout 5s "$IP" 2>/dev/null)
          echo "$nmap_output" | grep "53/tcp" | grep -q "open" && NMAP_OK=1
        fi

        if [ "$NC_OK" -eq 1 ] || [ "$TELNET_OK" -eq 1 ] || [ "$DIG_OK" -eq 1 ] || [ "$NMAP_OK" -eq 1 ]; then
          echo "DNS1_OK"
        else
          echo "DNS1_FALHA"
        fi
      register: dns1_test
      when: ssh_ok | default(false)
      changed_when: false

    - name: DNS | Testar conectividade DNS - IP 2
      ansible.builtin.raw: |
        IP="{{ dns_ips[datacenter][1] }}"
        NC_OK=0
        TELNET_OK=0
        DIG_OK=0
        NMAP_OK=0

        if command -v nc >/dev/null 2>&1 && nc -h 2>&1 | grep -q '\-z'; then
          nc -z -w2 "$IP" 53 >/dev/null 2>&1 && NC_OK=1
        elif command -v telnet >/dev/null 2>&1; then
          telnet_output=$(echo quit | telnet "$IP" 53 2>/dev/null)
          echo "$telnet_output" | grep -q "Escape character" && TELNET_OK=1
        fi

        if command -v dig >/dev/null 2>&1; then
          dig_output=$(dig @"$IP" google.com +time=2 +tries=1 2>/dev/null)
          echo "$dig_output" | grep -q "ANSWER SECTION" && DIG_OK=1
        fi

        if command -v nmap >/dev/null 2>&1; then
          nmap_output=$(nmap -p 53 -Pn --host-timeout 5s "$IP" 2>/dev/null)
          echo "$nmap_output" | grep "53/tcp" | grep -q "open" && NMAP_OK=1
        fi

        if [ "$NC_OK" -eq 1 ] || [ "$TELNET_OK" -eq 1 ] || [ "$DIG_OK" -eq 1 ] || [ "$NMAP_OK" -eq 1 ]; then
          echo "DNS2_OK"
        else
          echo "DNS2_FALHA"
        fi
      register: dns2_test
      when: ssh_ok | default(false)
      changed_when: false

    - name: DNS | Registrar status porta 53
      ansible.builtin.set_fact:
        dns1_ok: "{{ dns1_test.stdout is defined and 'DNS1_OK' in dns1_test.stdout }}"
        dns2_ok: "{{ dns2_test.stdout is defined and 'DNS2_OK' in dns2_test.stdout }}"
      when: ssh_ok | default(false)

    - name: DNS | Ler resolv.conf (somente leitura)
      ansible.builtin.raw: cat /etc/resolv.conf 2>/dev/null
      register: resolv_conf
      when: ssh_ok | default(false)
      changed_when: false

    - name: DNS | Verificar marcador ANSIBLE no resolv.conf
      ansible.builtin.raw: grep -q 'ANSIBLE' /etc/resolv.conf 2>/dev/null
      register: resolv_marker
      when: ssh_ok | default(false)
      changed_when: false
      failed_when: false

    - name: DNS | Validar nameservers Infoblox com correspondencia exata
      ansible.builtin.raw: |
        IP1="{{ dns_ips[datacenter][0] }}"
        IP2="{{ dns_ips[datacenter][1] }}"
        FOUND1=0
        FOUND2=0
        awk -v ip="$IP1" '$1 == "nameserver" && $2 == ip { found=1 } END { exit(found ? 0 : 1) }' /etc/resolv.conf && FOUND1=1
        awk -v ip="$IP2" '$1 == "nameserver" && $2 == ip { found=1 } END { exit(found ? 0 : 1) }' /etc/resolv.conf && FOUND2=1
        if [ "$FOUND1" -eq 1 ] && [ "$FOUND2" -eq 1 ]; then
          echo "RESOLV_OK"
        else
          echo "RESOLV_NAO_OK"
        fi
      register: resolv_check
      when: ssh_ok | default(false)
      changed_when: false

    - name: DNS | Registrar status do resolv.conf
      ansible.builtin.set_fact:
        resolv_conf_ok: "{{ resolv_check.stdout is defined and 'RESOLV_OK' in resolv_check.stdout }}"
        resolv_managed_by_ansible: "{{ resolv_marker.rc | default(1) == 0 }}"
      when: ssh_ok | default(false)

    # =========================================================================
    # ETAPA 3B: COLETAS ADICIONAIS - EXCLUSIVAS DA POS-VALIDACAO (READ-ONLY)
    # =========================================================================
    - name: POS | Ler atributos do resolv.conf (lsattr - somente leitura)
      ansible.builtin.raw: |
        if command -v lsattr >/dev/null 2>&1; then
          ATTRS=$(lsattr /etc/resolv.conf 2>/dev/null | awk '{print $1}')
          echo "ATTRS=$ATTRS"
          case "$ATTRS" in
            *i*) echo "IMMUT_YES" ;;
            *)   echo "IMMUT_NO" ;;
          esac
        else
          echo "LSATTR_NA"
        fi
      register: lsattr_result
      when:
        - modo == 'pos-validacao'
        - ssh_ok | default(false)
      changed_when: false

    - name: POS | Ler nsswitch.conf (linha hosts - somente leitura)
      ansible.builtin.raw: grep '^hosts:' /etc/nsswitch.conf 2>/dev/null || echo "NSSWITCH_SEM_HOSTS"
      register: nsswitch_result
      when:
        - modo == 'pos-validacao'
        - ssh_ok | default(false)
      changed_when: false

    - name: POS | Consolidar evidências da pós-validação
      ansible.builtin.set_fact:
        escopo_ok: "{{ ('Red Hat' in (so_detect.stdout | default(''))) or ('CentOS' in (so_detect.stdout | default(''))) or ('Oracle' in (so_detect.stdout | default(''))) or ('SunOS' in (so_detect.stdout | default(''))) }}"
        is_rhel: "{{ ('Red Hat' in (so_detect.stdout | default(''))) or ('CentOS' in (so_detect.stdout | default(''))) or ('Oracle' in (so_detect.stdout | default(''))) }}"
        is_sunos: "{{ 'SunOS' in (so_detect.stdout | default('')) }}"
        imutavel_ok: "{{ 'IMMUT_YES' in (lsattr_result.stdout | default('')) }}"
        nsswitch_line: "{{ (nsswitch_result.stdout | default('')) | trim }}"
        nsswitch_ok: "{{ ('dns' in (nsswitch_result.stdout | default(''))) and ('files' in (nsswitch_result.stdout | default(''))) }}"
        dns_encontrado: "{{ resolv_conf.stdout_lines | default([]) | select('search', '^nameserver') | map('replace', 'nameserver ', '') | map('trim') | list | join(', ') }}"
      when:
        - modo == 'pos-validacao'
        - ssh_ok | default(false)

    # =========================================================================
    # ETAPA 3C: DEPLOY - ALTERAÇÃO EFETIVA DO DNS (EXCLUSIVA DO MODO deploy)
    # Só altera hosts que o pre-check classificaria como "OK PARA ALTERAÇÃO":
    #   SSH ok + SO em escopo + resolv.conf ainda NÃO correto + sem marcador
    #   ANSIBLE + porta 53 ok nos dois DNS do datacenter.
    # A alteração em si é feita pelos roles existentes (rhel_* / sunos_*).
    # =========================================================================
    - name: DEPLOY | Definir família do SO e nameservers atuais
      ansible.builtin.set_fact:
        deploy_familia: >-
          {%- if not (ssh_ok | default(false)) -%}
            none
          {%- elif (so_detectado | default('')) in ['RHEL', 'CentOS', 'Oracle Linux'] -%}
            rhel
          {%- elif (so_detectado | default('')) == 'SunOS' -%}
            sunos
          {%- else -%}
            none
          {%- endif -%}
        deploy_nameservers_antes: "{{ (resolv_conf.stdout_lines | default([]) | select('search', '^\\s*nameserver\\s+') | list | join(' | ')) | trim }}"
      when: modo == 'deploy'

    - name: DEPLOY | Definir elegibilidade e role a executar
      ansible.builtin.set_fact:
        deploy_elegivel: "{{ (ssh_ok | default(false)) and (deploy_familia != 'none') and not (resolv_conf_ok | default(false)) and not (resolv_managed_by_ansible | default(false)) and (dns1_ok | default(false)) and (dns2_ok | default(false)) }}"
        deploy_role: "{{ deploy_familia ~ '_' ~ dc_role_sufixo[datacenter] }}"
      when: modo == 'deploy'

    - name: DEPLOY | Executar role de alteração (rhel_* / sunos_* do datacenter)
      ansible.builtin.include_role:
        name: "{{ deploy_role }}"
      when:
        - modo == 'deploy'
        - deploy_elegivel | default(false) | bool

    # Pós-verificação (somente leitura): mesma lógica da pos-validacao.
    - name: DEPLOY | Pós-verificação do host alterado (somente leitura)
      ansible.builtin.raw: |
        IP1="{{ dns_ips[datacenter][0] }}"
        IP2="{{ dns_ips[datacenter][1] }}"
        FOUND1=0
        FOUND2=0
        awk -v ip="$IP1" '$1 == "nameserver" && $2 == ip { found=1 } END { exit(found ? 0 : 1) }' /etc/resolv.conf && FOUND1=1
        awk -v ip="$IP2" '$1 == "nameserver" && $2 == ip { found=1 } END { exit(found ? 0 : 1) }' /etc/resolv.conf && FOUND2=1
        if [ "$FOUND1" -eq 1 ] && [ "$FOUND2" -eq 1 ]; then
          echo "RESOLV_OK"
        else
          echo "RESOLV_NAO_OK"
        fi

        if command -v lsattr >/dev/null 2>&1; then
          ATTRS=$(lsattr /etc/resolv.conf 2>/dev/null | awk '{print $1}')
          case "$ATTRS" in
            *i*) echo "IMMUT_YES" ;;
            *)   echo "IMMUT_NO" ;;
          esac
        else
          echo "LSATTR_NA"
        fi

        LINE=$(grep '^hosts:' /etc/nsswitch.conf 2>/dev/null)
        echo "NSSWITCH=$LINE"
        if echo "$LINE" | grep files >/dev/null 2>&1 && echo "$LINE" | grep dns >/dev/null 2>&1; then
          echo "NSS_OK"
        else
          echo "NSS_NAO_OK"
        fi

        cat /etc/resolv.conf 2>/dev/null
      register: deploy_pos
      when:
        - modo == 'deploy'
        - deploy_elegivel | default(false) | bool
      changed_when: false

    - name: DEPLOY | Registrar pós-verificação
      ansible.builtin.set_fact:
        dep_resolv_ok: "{{ 'RESOLV_OK' in (deploy_pos.stdout | default('')) }}"
        dep_imutavel_ok: "{{ 'IMMUT_YES' in (deploy_pos.stdout | default('')) }}"
        dep_nsswitch_ok: "{{ 'NSS_OK' in (deploy_pos.stdout | default('')) }}"
        dep_nsswitch_line: "{{ ((deploy_pos.stdout_lines | default([])) | select('match', '^NSSWITCH=') | list | first | default('NSSWITCH=')) | regex_replace('^NSSWITCH=', '') | trim }}"
        deploy_nameservers_depois: "{{ ((deploy_pos.stdout_lines | default([])) | select('search', '^\\s*nameserver\\s+') | list | join(' | ')) | trim }}"
      when:
        - modo == 'deploy'
        - deploy_elegivel | default(false) | bool

    - name: DEPLOY | Consolidar resultado da pós-verificação
      ansible.builtin.set_fact:
        deploy_pos_ok: "{{ (dep_resolv_ok | default(false)) and ((deploy_familia != 'rhel') or (dep_imutavel_ok | default(false))) and ((deploy_familia != 'sunos') or (dep_nsswitch_ok | default(false))) }}"
      when:
        - modo == 'deploy'
        - deploy_elegivel | default(false) | bool

    # -------------------------------------------------------------------------
    # ETAPA 4A: CLASSIFICAR RESULTADO - MODO VALIDATION
    # Classificacao operacional: somente o que interessa para o deploy.
    # -------------------------------------------------------------------------
    - name: RESULTADO | Classificar host (pre-check)
      ansible.builtin.set_fact:
        resultado_host:
          ip: "{{ inventory_hostname }}"
          so: "{{ so_detectado | default('Desconhecido') }}"
          datacenter: "{{ datacenter }}"
          resolv_conf_status: >-
            {%- if resolv_conf_ok | default(false) -%}
              RESOLV.CONF JÁ CORRETO
            {%- else -%}
              DNS NAO CONFIGURADO
            {%- endif -%}
          classificacao: >-
            {%- if not (ssh_ok | default(false)) -%}
              FAZER MANUALMENTE
            {%- elif resolv_conf_ok | default(false) -%}
              RESOLV.CONF JÁ CORRETO
            {%- elif (so_detectado | default('')) not in ['RHEL', 'CentOS', 'Oracle Linux', 'SunOS'] -%}
              FAZER MANUALMENTE
            {%- elif resolv_managed_by_ansible | default(false) -%}
              FAZER MANUALMENTE
            {%- elif (dns1_ok | default(false)) and (dns2_ok | default(false)) -%}
              OK PARA ALTERAÇÃO
            {%- else -%}
              FAZER MANUALMENTE
            {%- endif -%}
          status: >-
            {%- if not (ssh_ok | default(false)) -%}FALHA
            {%- elif resolv_conf_ok | default(false) -%}OK
            {%- elif (dns1_ok | default(false)) and (dns2_ok | default(false)) -%}OK
            {%- else -%}FALHA
            {%- endif -%}
          tipo_erro: >-
            {%- set erro_msg = ssh_erro_msg | default('') -%}
            {%- if not (ssh_ok | default(false)) -%}
              {%- if 'Permission denied' in erro_msg -%}PERMISSION_DENIED
              {%- elif 'Connection refused' in erro_msg -%}FIREWALL_22
              {%- elif 'Connection timed out' in erro_msg or 'No route to host' in erro_msg -%}FIREWALL_22
              {%- elif 'Timer expired' in erro_msg or 'timed out' in erro_msg -%}TIMEOUT
              {%- else -%}FALHA_SSH
              {%- endif -%}
            {%- elif resolv_conf_ok | default(false) -%}NENHUM
            {%- elif resolv_managed_by_ansible | default(false) -%}MARCADOR_ANSIBLE
            {%- elif (so_detectado | default('')) not in ['RHEL', 'CentOS', 'Oracle Linux', 'SunOS'] -%}SO_FORA_ESCOPO
            {%- elif not (dns1_ok | default(false)) or not (dns2_ok | default(false)) -%}FIREWALL_53
            {%- else -%}NENHUM
            {%- endif -%}
          nameservers_atuais: "{{ (resolv_conf.stdout_lines | default([]) | select('search', '^\\s*nameserver\\s+') | list | join(' | ')) | trim }}"
          mensagem: >-
            {%- set erro_msg = ssh_erro_msg | default('') -%}
            {%- if not (ssh_ok | default(false)) -%}
              Sem acesso SSH{% if erro_msg %} - {{ erro_msg | regex_replace('\n', ' ') | truncate(120) }}{% endif %}
            {%- elif resolv_conf_ok | default(false) -%}
              DNS Infoblox ja configurado no /etc/resolv.conf
            {%- elif resolv_managed_by_ansible | default(false) -%}
              Arquivo possui marcador ANSIBLE, mas DNS esperado nao esta configurado
            {%- elif (so_detectado | default('')) not in ['RHEL', 'CentOS', 'Oracle Linux', 'SunOS'] -%}
              Sistema operacional fora do escopo do deploy
            {%- elif not (dns1_ok | default(false)) or not (dns2_ok | default(false)) -%}
              Comunicacao com um ou ambos os DNS Infoblox na porta 53 indisponivel
            {%- else -%}
              DNS ainda nao configurado; servidor apto para alteracao automatica
            {%- endif -%}
      when: modo == 'validation'

    # -------------------------------------------------------------------------
    # ETAPA 4B: CLASSIFICAR RESULTADO - MODO POS-VALIDACAO (fiel aos roles)
    # -------------------------------------------------------------------------
    - name: RESULTADO | Classificar host (pos-validacao)
      ansible.builtin.set_fact:
        resultado_host:
          ip: "{{ inventory_hostname }}"
          datacenter: "{{ datacenter }}"
          so: "{{ so_detectado | default('Desconhecido') }}"
          escopo: >-
            {%- if not (ssh_ok | default(false)) -%}
              INDETERMINADO
            {%- elif escopo_ok | default(false) -%}
              EM ESCOPO
            {%- else -%}
              FORA DE ESCOPO
            {%- endif -%}
          dns_esperado: "{{ dns_ips[datacenter] | join(', ') }}"
          dns_encontrado: "{{ dns_encontrado | default('') }}"
          teste_dns1: >-
            {%- if not (ssh_ok | default(false)) -%}
              N/A
            {%- elif dns1_ok | default(false) -%}
              OK
            {%- else -%}
              FALHA
            {%- endif -%}
          teste_dns2: >-
            {%- if not (ssh_ok | default(false)) -%}
              N/A
            {%- elif dns2_ok | default(false) -%}
              OK
            {%- else -%}
              FALHA
            {%- endif -%}
          resolv_conf: >-
            {%- if not (ssh_ok | default(false)) -%}
              N/A
            {%- elif resolv_conf_ok | default(false) -%}
              DNS esperado presente
            {%- else -%}
              DNS esperado ausente
            {%- endif -%}
          nsswitch: >-
            {%- if not (is_sunos | default(false)) -%}
              N/A
            {%- elif nsswitch_ok | default(false) -%}
              hosts: files dns
            {%- else -%}
              {{ nsswitch_line | default('ausente') }}
            {%- endif -%}
          imutabilidade: >-
            {%- if not (is_rhel | default(false)) -%}
              N/A
            {%- elif imutavel_ok | default(false) -%}
              IMUTAVEL
            {%- else -%}
              NAO IMUTAVEL
            {%- endif -%}
          status: >-
            {%- if not (ssh_ok | default(false)) -%}
              ERRO DE CONECTIVIDADE
            {%- elif not (escopo_ok | default(false)) -%}
              FORA DE ESCOPO
            {%- elif not (resolv_conf_ok | default(false)) -%}
              FALHA
            {%- elif not (dns1_ok | default(false)) or not (dns2_ok | default(false)) -%}
              FALHA
            {%- elif (is_rhel | default(false)) and not (imutavel_ok | default(false)) -%}
              FALHA
            {%- elif (is_sunos | default(false)) and not (nsswitch_ok | default(false)) -%}
              FALHA
            {%- else -%}
              SUCESSO
            {%- endif -%}
          motivo: >-
            {%- set erro_msg = ssh_erro_msg | default('') -%}
            {%- if not (ssh_ok | default(false)) -%}
              {%- if 'Permission denied' in erro_msg -%}
                SSH indisponível - Permission Denied
              {%- elif 'Connection refused' in erro_msg -%}
                SSH indisponível - Porta 22 recusada (Connection refused)
              {%- elif 'timed out' in erro_msg or 'No route to host' in erro_msg or 'Timer expired' in erro_msg -%}
                SSH indisponível - Timeout ou sem rota
              {%- else -%}
                SSH indisponível
              {%- endif -%}
            {%- elif not (escopo_ok | default(false)) -%}
