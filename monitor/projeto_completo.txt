---
- name: Atualiza e cria alertas de certificado no Zabbix baseado em chamados de GA
  hosts: localhost
  gather_facts: true
  environment:
    TZ: "America/Sao_Paulo"

  vars:
    # ----------------------------------------------------------
    # Resumos (summary) que diferenciam os dois tipos de chamado.
    # Usados na JQL e no filtro local da role descoberta.
    # ----------------------------------------------------------
    #jira_resumo_atualizacao: "Atualização de Alerta de Certificado - Zabbix"
    jira_resumo_atualizacao: "Atualização de alerta"
    jira_resumo_criacao: "Criação de Alerta de Certificado - Zabbix"


    # ----------------------------------------------------------
    # Zabbix: host e padrão de chave dos itens de certificado.
    # Chave: zbx_valida_data_certificado["host","Tipo","DD/MM/AAAA"]
    #
    # zabbix_chave_regex  -> extrai (host, tipo, data) de uma chave.
    # zabbix_data_regex   -> localiza somente a data dentro da chave.
    # (regex em aspas simples do YAML: a barra invertida é literal)
    # ----------------------------------------------------------
    zabbix_host_certificados: "CERTIFICADOS_CLARO_BRASIL"
    zabbix_chave_certificado: "zbx_valida_data_certificado"
    zabbix_chave_regex: '^zbx_valida_data_certificado\[\s*"([^"]*)"\s*,\s*"([^"]*)"\s*,\s*"([0-9]{2}/[0-9]{2}/[0-9]{4})"\s*\]$'
    zabbix_data_regex: '[0-9]{2}/[0-9]{2}/[0-9]{4}(?="\s*\]$)'

    # URL da API (antes montada com set_fact dentro das roles).
    zabbix_api_url: >-
      {{ 'https' if zabbix_use_ssl | bool else 'http' }}://{{ zabbix_host }}{{ (':' ~ zabbix_port) if zabbix_port | int not in [80, 443] else '' }}{{ ('/' ~ zabbix_url_path) if zabbix_url_path | length > 0 else '' }}/api_jsonrpc.php

    # ----------------------------------------------------------
    # Segurança: por padrão NADA é gravado no Zabbix (simulação).
    # Para gravar de verdade, informe aplicar_alteracoes: true
    # (Survey do AAP ou extra vars).
    # ----------------------------------------------------------
    zabbix_aplicar_alteracoes: "{{ aplicar_alteracoes | default(false) | bool }}"

    # ----------------------------------------------------------
    # Survey (opcional): número do chamado, ex.: PDST-123456.
    # Em branco -> busca todos os chamados em aberto (comportamento
    # de sempre). Preenchido -> processa SOMENTE esse chamado.
    # ----------------------------------------------------------
    jira_chamado: "{{ chamado | default('') | string | trim | upper }}"

  pre_tasks:
    # Falha antes de qualquer alteração no Zabbix, para não alterar
    # nada e só então descobrir que o relatório não tem destinatário.
    - name: Validar o destinatário do relatório (variável email da Survey)
      ansible.builtin.assert:
        that:
          - email is defined
          - email | string | trim | length > 0
          - "'@' in (email | string)"
        fail_msg: >-
          Informe a variável "email" na Survey do AAP com o endereço que
          receberá o relatório (ex.: usuario@empresa.com).
        quiet: true

    # Formato do chamado (evita montar uma JQL inválida ou com texto estranho).
    - name: Validar o número do chamado (variável chamado da Survey, opcional)
      ansible.builtin.assert:
        that:
          - jira_chamado | length == 0 or jira_chamado is match('^[A-Z][A-Z0-9_]*-[0-9]+$')
        fail_msg: >-
          O chamado informado ("{{ jira_chamado }}") não tem o formato esperado.
          Use algo como PDST-123456, ou deixe em branco para buscar todos.
        quiet: true

    - name: Validar que o Zabbix está habilitado
      ansible.builtin.assert:
        that:
          - zabbix_habilitado | default(true) | bool
        fail_msg: >-
          Este playbook só trabalha com o Zabbix; zabbix_habilitado
          não pode ser false.
        quiet: true

  roles:
    # Cria o host virtual usado pelas tasks do Zabbix
    - role: inventory

    # Procura os chamados no Jira e separa em:
    #   certificados_para_atualizar / certificados_para_criar
    - role: descoberta

    # Altera a validade (chave) de itens já cadastrados
    - role: atualiza
      when: certificados_para_atualizar | length > 0

    # Cadastra itens novos
    - role: cria
      when: certificados_para_criar | length > 0

    # Gera o CSV e envia para o e-mail informado na Survey
    # (tasks/certificados.yml)
    - role: relatorios
      tasks_from: certificados

# ============================================================
# Role atualiza: altera a validade (data dentro da key) de itens
# JÁ cadastrados no host CERTIFICADOS_CLARO_BRASIL.
#
# Entrada (role descoberta):
#   certificados_para_atualizar:
#     - ticket: GA-123
#       operacao: ATUALIZACAO
#       host: netsmsmfa.claro.com.br
#       tipo: Externo
#       validade: "30/03/2027"
#
# Saída (consumida pela role relatorios):
#   resultado_atualizacao: uma linha por certificado, com
#   operacao, ticket, host, tipo, itemid, validade_anterior,
#   validade_solicitada, status e mensagem.
#
# Status possíveis:
#   ATUALIZADO      API confirmou a alteração
#   SIMULADO        aplicar_alteracoes = false (nada foi gravado)
#   JA_ATUALIZADO   o item já está com a validade pedida
#   NAO_ENCONTRADO  nenhum item para host + tipo
#   AMBIGUO         mais de um item para host + tipo e não há exatamente
#                   um Enabled (não altera). Se houver exatamente um
#                   Enabled entre vários, é ele que é atualizado.
#   CONFLITO        chamados pedem validades diferentes para o
#                   mesmo certificado (não altera)
#   TIPO_INVALIDO   descrição do chamado sem Externo/Interno
#   ERRO            a API retornou erro ou não respondeu
#
# Fluxo: 1 consulta em lote -> plano por certificado (sem
# escrita) -> item.update só dos PENDENTES -> resultado final.
# ============================================================

# ------------------------------------------------------------
# 1. Buscar de uma vez todos os itens de certificado do host
#    (uma chamada em vez de uma por certificado).
#    JSON-RPC devolve HTTP 200 mesmo com erro: valida "error".
# ------------------------------------------------------------
- name: Buscar itens de certificado no Zabbix
  ansible.builtin.uri:
    url: "{{ zabbix_api_url }}"
    method: POST
    body_format: json
    headers:
      Content-Type: application/json-rpc
    body:
      jsonrpc: "2.0"
      method: "item.get"
      params:
        host: "{{ zabbix_host_certificados }}"
        search:
          key_: "{{ zabbix_chave_certificado }}"
        output:
          - itemid
          - name
          - key_
          - status
      auth: "{{ zabbix_api_token }}"
      id: 1
    validate_certs: "{{ zabbix_validate_certs }}"
  delegate_to: localhost
  register: zbx_itens
  failed_when: >-
    zbx_itens.json is not defined
    or zbx_itens.json.error is defined
    or zbx_itens.json.result is not defined

# ------------------------------------------------------------
# 2. Separar host / tipo / data de cada key existente.
#    Só entram itens cuja key segue o padrão
#    zbx_valida_data_certificado["host","Tipo","DD/MM/AAAA"].
#    host e tipo ficam em minúsculas para comparação.
# ------------------------------------------------------------
- name: Interpretar a key dos itens existentes
  ansible.builtin.set_fact:
    zbx_certs: >-
      {%- set r = namespace(lista=[]) -%}
      {%- for i in zbx_itens.json.result -%}
      {%- set p = i.key_ | regex_findall(zabbix_chave_regex) -%}
      {%- if p -%}
      {%- set r.lista = r.lista + [{'itemid': i.itemid, 'name': i.name, 'key_': i.key_, 'status': i.status | string, 'host': p[0][0] | lower, 'tipo': p[0][1] | lower, 'data': p[0][2]}] -%}
      {%- endif -%}
      {%- endfor -%}
      {{ r.lista }}

# ------------------------------------------------------------
# 3. Planejar: decide o que fazer com cada certificado, SEM
#    alterar nada. A ordem do "status" abaixo é a ordem de
#    prioridade das regras. O debug serve só para registrar uma
#    linha por certificado (resultado em zbx_plano_loop).
# ------------------------------------------------------------
- name: Planejar atualizações (somente leitura)
  ansible.builtin.debug:
    msg:
      operacao: "ATUALIZACAO"
      ticket: "{{ item.ticket }}"
      host: "{{ item.host }}"
      tipo: "{{ item.tipo }}"
      itemid: "{{ encontrados[0].itemid if encontrados | length == 1 else '' }}"
      validade_anterior: "{{ encontrados[0].data if encontrados | length == 1 else '' }}"
      validade_solicitada: "{{ item.validade }}"
      status: "{{ status }}"
      mensagem: "{{ mensagem }}"
      nova_chave: "{{ nova_chave }}"
  vars:
    # itens do Zabbix com o mesmo host + tipo (comparação exata na key)
    candidatos: >-
      {{ zbx_certs
         | selectattr('host', 'equalto', item.host | lower)
         | selectattr('tipo', 'equalto', item.tipo | lower)
         | list }}
    # havendo mais de um candidato, desempata ficando só com o(s) item(ns)
    # Enabled (na API: status "0"; "1" = Disabled)
    encontrados: >-
      {{ candidatos if candidatos | length <= 1
         else candidatos | selectattr('status', 'equalto', '0') | list }}
    # validades diferentes pedidas para o mesmo host + tipo (todos os chamados)
    datas_pedidas: >-
      {{ certificados_para_atualizar
         | selectattr('host', 'equalto', item.host)
         | selectattr('tipo', 'equalto', item.tipo)
         | map(attribute='validade')
         | unique
         | list }}
    status: >-
      {{ 'TIPO_INVALIDO' if (item.tipo | lower) not in ['externo', 'interno']
         else 'CONFLITO' if datas_pedidas | length > 1
         else 'NAO_ENCONTRADO' if candidatos | length == 0
         else 'AMBIGUO' if encontrados | length != 1
         else 'JA_ATUALIZADO' if encontrados[0].data == item.validade
         else 'PENDENTE' }}
    mensagem: >-
      {{ 'Tipo não identificado na descrição do chamado (esperado Externo ou Interno).' if status == 'TIPO_INVALIDO'
         else 'Chamados pedem validades diferentes para o mesmo certificado (' ~ (datas_pedidas | join(', ')) ~ '). Nada foi alterado.' if status == 'CONFLITO'
         else 'Nenhum item encontrado no Zabbix para este host e tipo. Para cadastrar, use um chamado de criação.' if status == 'NAO_ENCONTRADO'
         else 'Mais de um item corresponde (itemids: ' ~ (candidatos | map(attribute='itemid') | join(', ')) ~ ') e não há exatamente um Enabled. Nada foi alterado.' if status == 'AMBIGUO'
         else 'O item já está com a validade solicitada.' if status == 'JA_ATUALIZADO'
         else 'Aguardando atualização.' }}
    # só substitui a data; o restante da key é mantido como está
    nova_chave: >-
      {{ encontrados[0].key_ | regex_replace(zabbix_data_regex, item.validade) if status == 'PENDENTE' else '' }}
  loop: "{{ certificados_para_atualizar }}"
  loop_control:
    label: "{{ item.host }} ({{ item.tipo }})"
  register: zbx_plano_loop

- name: Guardar o plano
  ansible.builtin.set_fact:
    zbx_plano: "{{ zbx_plano_loop.results | map(attribute='msg') | list }}"

# ------------------------------------------------------------
# 4. Atualizar. Só vai para a API o que ficou PENDENTE e somente
#    com aplicar_alteracoes = true. Enviamos apenas itemid + key_,
#    então os demais atributos do item não são tocados.
#    failed_when: false -> um erro não interrompe os outros itens;
#    o resultado real de cada chamada é avaliado no passo 5.
# ------------------------------------------------------------
- name: Atualizar a key dos itens no Zabbix
  ansible.builtin.uri:
    url: "{{ zabbix_api_url }}"
    method: POST
    body_format: json
    headers:
      Content-Type: application/json-rpc
    body:
      jsonrpc: "2.0"
      method: "item.update"
      params:
        itemid: "{{ item.itemid }}"
        key_: "{{ item.nova_chave }}"
      auth: "{{ zabbix_api_token }}"
      id: 1
    validate_certs: "{{ zabbix_validate_certs }}"
  delegate_to: localhost
  loop: "{{ zbx_plano }}"
  loop_control:
    label: "{{ item.host }} ({{ item.tipo }})"
  when: item.status == 'PENDENTE' and zabbix_aplicar_alteracoes | bool
  register: zbx_update
  failed_when: false

# ------------------------------------------------------------
# 5. Resultado final. Sucesso = HTTP 200 + sem "error" no JSON-RPC
#    + itemid confirmado em result.itemids. Qualquer outra coisa
#    vira ERRO com a mensagem da API (ou do transporte).
# ------------------------------------------------------------
- name: Consolidar resultado da atualização
  ansible.builtin.debug:
    msg:
      operacao: "{{ linha.operacao }}"
      ticket: "{{ linha.ticket }}"
      host: "{{ linha.host }}"
      tipo: "{{ linha.tipo }}"
      itemid: "{{ linha.itemid }}"
      validade_anterior: "{{ linha.validade_anterior }}"
      validade_solicitada: "{{ linha.validade_solicitada }}"
      status: "{{ status }}"
      mensagem: "{{ mensagem }}"
  vars:
    linha: "{{ item.0 }}"
    resp: "{{ item.1 }}"
    api_ok: >-
      {{ resp.json is defined
         and resp.json.error is not defined
         and linha.itemid in (resp.json.result.itemids | default([])) }}
    api_erro: >-
      {{ (resp.json.error.message ~ ' - ' ~ (resp.json.error.data | default('')))
         if resp.json is defined and resp.json.error is defined
         else (resp.msg | default('Sem resposta válida da API.')) }}
    status: >-
      {{ linha.status if linha.status != 'PENDENTE'
         else 'SIMULADO' if resp.skipped | default(false)
         else 'ATUALIZADO' if api_ok
         else 'ERRO' }}
    mensagem: >-
      {{ linha.mensagem if linha.status != 'PENDENTE'
         else 'Simulação (aplicar_alteracoes=false): a key passaria para ' ~ linha.nova_chave if status == 'SIMULADO'
         else 'Validade alterada de ' ~ linha.validade_anterior ~ ' para ' ~ linha.validade_solicitada ~ '.' if status == 'ATUALIZADO'
         else api_erro }}
  loop: "{{ zbx_plano | zip(zbx_update.results) | list }}"
  loop_control:
    label: "{{ item.0.host }} ({{ item.0.tipo }})"
  register: zbx_resultado_loop

- name: Guardar o resultado para o relatório
  ansible.builtin.set_fact:
    resultado_atualizacao: "{{ zbx_resultado_loop.results | map(attribute='msg') | list }}"

# ============================================================
# Role cria: cadastra itens NOVOS de certificado no host
# CERTIFICADOS_CLARO_BRASIL, sem duplicar os que já existem.
#
# Entrada (role descoberta):
#   certificados_para_criar:
#     - ticket: GA-456
#       operacao: CRIACAO
#       host: novo.claro.com.br
#       tipo: Externo
#       validade: "20/08/2027"
#
# Saída (consumida pela role relatorios):
#   resultado_criacao (mesmas colunas de resultado_atualizacao).
#
# Status possíveis:
#   CRIADO         API confirmou a criação (itemid novo na linha)
#   SIMULADO       aplicar_alteracoes = false (nada foi gravado)
#   JA_EXISTE      já há item para host + tipo (não cria; para mudar
#                  a data use chamado de atualização)
#   SEM_MODELO     não há item-modelo Enabled do mesmo tipo no host, ou o nome
#                  dele não permite derivar o nome do novo item
#   CONFLITO       chamados pedem validades diferentes para o mesmo
#                  certificado (não cria)
#   TIPO_INVALIDO  descrição do chamado sem Externo/Interno
#   ERRO           a API retornou erro ou não respondeu
#
# Como o item novo é montado (nada é inventado):
#   - key_  : a key do item-modelo, trocando host e data;
#   - name  : o nome do item-modelo, trocando host e data;
#   - hostid: obtido por host.get do host de certificados;
#   - demais atributos: copiados do item-modelo (cria_campos_modelo).
# ============================================================

# ------------------------------------------------------------
# 1. hostid do host de certificados (necessário no item.create).
# ------------------------------------------------------------
- name: Buscar o hostid do host de certificados
  ansible.builtin.uri:
    url: "{{ zabbix_api_url }}"
    method: POST
    body_format: json
    headers:
      Content-Type: application/json-rpc
    body:
      jsonrpc: "2.0"
      method: "host.get"
      params:
        filter:
          host:
            - "{{ zabbix_host_certificados }}"
        output:
          - hostid
          - host
      auth: "{{ zabbix_api_token }}"
      id: 1
    validate_certs: "{{ zabbix_validate_certs }}"
  delegate_to: localhost
  register: zbx_host
  failed_when: >-
    zbx_host.json is not defined
    or zbx_host.json.error is defined
    or zbx_host.json.result is not defined
    or zbx_host.json.result | length != 1

- name: Guardar o hostid
  ansible.builtin.set_fact:
    zbx_hostid: "{{ zbx_host.json.result[0].hostid }}"

# ------------------------------------------------------------
# 2. Itens de certificado já existentes, com TODOS os atributos
#    (output extend), tags e pré-processamento: servem para
#    detectar duplicatas e como modelo.
# ------------------------------------------------------------
- name: Buscar itens de certificado no Zabbix
  ansible.builtin.uri:
    url: "{{ zabbix_api_url }}"
    method: POST
    body_format: json
    headers:
      Content-Type: application/json-rpc
    body:
      jsonrpc: "2.0"
      method: "item.get"
      params:
        hostids: "{{ zbx_hostid }}"
        search:
          key_: "{{ zabbix_chave_certificado }}"
        output: "extend"
        selectTags: "extend"
        selectPreprocessing: "extend"
        sortfield: "itemid"
      auth: "{{ zabbix_api_token }}"
      id: 1
    validate_certs: "{{ zabbix_validate_certs }}"
  delegate_to: localhost
  register: zbx_itens
  failed_when: >-
    zbx_itens.json is not defined
    or zbx_itens.json.error is defined
    or zbx_itens.json.result is not defined

# host_orig guarda o host exatamente como está na key (com maiúsculas
# e minúsculas), usado para trocar o host no nome e na key do modelo.
- name: Interpretar a key dos itens existentes
  ansible.builtin.set_fact:
    zbx_certs: >-
      {%- set r = namespace(lista=[]) -%}
      {%- for i in zbx_itens.json.result -%}
      {%- set p = i.key_ | regex_findall(zabbix_chave_regex) -%}
      {%- if p -%}
      {%- set r.lista = r.lista + [{'itemid': i.itemid, 'name': i.name, 'key_': i.key_, 'host': p[0][0] | lower, 'host_orig': p[0][0], 'tipo': p[0][1] | lower, 'data': p[0][2], 'item': i}] -%}
      {%- endif -%}
      {%- endfor -%}
      {{ r.lista }}

# ------------------------------------------------------------
# 3. Planejar: decide o que fazer com cada certificado, SEM
#    criar nada. Ordem do "status" = prioridade das regras.
#    O payload mostrado é exatamente o que iria no item.create.
# ------------------------------------------------------------
- name: Planejar criações (somente leitura)
  ansible.builtin.debug:
    msg:
      operacao: "CRIACAO"
      ticket: "{{ item.ticket }}"
      host: "{{ item.host }}"
      tipo: "{{ item.tipo }}"
      itemid: "{{ existentes[0].itemid if existentes | length > 0 else '' }}"
      validade_anterior: "{{ existentes[0].data if existentes | length > 0 else '' }}"
      validade_solicitada: "{{ item.validade }}"
      status: "{{ status }}"
      mensagem: "{{ mensagem }}"
      payload: "{{ payload }}"
  vars:
    # itens que já existem para o mesmo host + tipo (qualquer data)
    existentes: >-
      {{ zbx_certs
         | selectattr('host', 'equalto', item.host | lower)
         | selectattr('tipo', 'equalto', item.tipo | lower)
         | list }}
    # item-modelo: primeiro item Enabled (menor itemid) do mesmo tipo.
    # Só Enabled, porque o status do modelo é copiado para o item novo.
    modelo: >-
      {{ zbx_certs
         | selectattr('tipo', 'equalto', item.tipo | lower)
         | selectattr('item.status', 'equalto', '0')
         | first
         | default({}) }}
    # o nome do modelo precisa conter o host para podermos trocá-lo
    modelo_ok: "{{ modelo | length > 0 and modelo.host_orig in modelo.item.name }}"
    datas_pedidas: >-
      {{ certificados_para_criar
         | selectattr('host', 'equalto', item.host)
         | selectattr('tipo', 'equalto', item.tipo)
         | map(attribute='validade')
         | unique
         | list }}
    status: >-
      {{ 'TIPO_INVALIDO' if (item.tipo | lower) not in ['externo', 'interno']
         else 'CONFLITO' if datas_pedidas | length > 1
         else 'JA_EXISTE' if existentes | length > 0
         else 'SEM_MODELO' if not modelo_ok
         else 'PENDENTE' }}
    mensagem: >-
      {{ 'Tipo não identificado na descrição do chamado (esperado Externo ou Interno).' if status == 'TIPO_INVALIDO'
         else 'Chamados pedem validades diferentes para o mesmo certificado (' ~ (datas_pedidas | join(', ')) ~ '). Nada foi criado.' if status == 'CONFLITO'
         else 'Já existe item para este host e tipo (itemid ' ~ existentes[0].itemid ~ ', validade ' ~ existentes[0].data ~ '). Nada foi criado; para mudar a data use um chamado de atualização.' if status == 'JA_EXISTE'
         else 'Sem item-modelo Enabled do tipo ' ~ item.tipo ~ ' no host, ou o nome do modelo não contém o host. Nada foi criado.' if status == 'SEM_MODELO'
         else 'Aguardando criação.' }}
    nome: >-
      {{ modelo.item.name | replace(modelo.host_orig, item.host) | replace(modelo.data, item.validade) if status == 'PENDENTE' else '' }}
    # key do modelo com o host e a data trocados (mantém o formato da key)
    chave: >-
      {{ modelo.key_ | replace('"' ~ modelo.host_orig ~ '"', '"' ~ item.host ~ '"') | regex_replace(zabbix_data_regex, item.validade) if status == 'PENDENTE' else '' }}
    # atributos do modelo (somente os de cria_campos_modelo, e só se preenchidos)
    campos_modelo: >-
      {{ modelo.item | dict2items
         | selectattr('key', 'in', cria_campos_modelo)
         | rejectattr('value', 'in', ['', []])
         | items2dict }}
    payload: >-
      {{ {'hostid': zbx_hostid, 'name': nome, 'key_': chave} | combine(campos_modelo) if status == 'PENDENTE' else {} }}
  loop: "{{ certificados_para_criar }}"
  loop_control:
    label: "{{ item.host }} ({{ item.tipo }})"
  register: zbx_plano_loop

- name: Guardar o plano
  ansible.builtin.set_fact:
    zbx_plano: "{{ zbx_plano_loop.results | map(attribute='msg') | list }}"

# ------------------------------------------------------------
# 4. Criar. Só vai para a API o que ficou PENDENTE e somente com
#    aplicar_alteracoes = true. failed_when: false -> um erro não
#    interrompe os demais; o resultado real é avaliado no passo 5.
# ------------------------------------------------------------
- name: Criar os itens no Zabbix
  ansible.builtin.uri:
    url: "{{ zabbix_api_url }}"
    method: POST
    body_format: json
    headers:
      Content-Type: application/json-rpc
    body:
      jsonrpc: "2.0"
      method: "item.create"
      params: "{{ item.payload }}"
      auth: "{{ zabbix_api_token }}"
      id: 1
    validate_certs: "{{ zabbix_validate_certs }}"
  delegate_to: localhost
  loop: "{{ zbx_plano }}"
  loop_control:
    label: "{{ item.host }} ({{ item.tipo }})"
  when: item.status == 'PENDENTE' and zabbix_aplicar_alteracoes | bool
  register: zbx_create
  failed_when: false

# ------------------------------------------------------------
# 5. Resultado final. Sucesso = HTTP 200 + sem "error" no JSON-RPC
#    + um itemid em result.itemids. Qualquer outra coisa vira ERRO
#    com a mensagem da API (ou do transporte).
# ------------------------------------------------------------
- name: Consolidar resultado da criação
  ansible.builtin.debug:
    msg:
      operacao: "{{ linha.operacao }}"
      ticket: "{{ linha.ticket }}"
      host: "{{ linha.host }}"
      tipo: "{{ linha.tipo }}"
      itemid: "{{ novo_itemid if status == 'CRIADO' else linha.itemid }}"
      validade_anterior: "{{ linha.validade_anterior }}"
      validade_solicitada: "{{ linha.validade_solicitada }}"
      status: "{{ status }}"
      mensagem: "{{ mensagem }}"
  vars:
    linha: "{{ item.0 }}"
    resp: "{{ item.1 }}"
    novo_itemid: "{{ resp.json.result.itemids[0] | default('') if resp.json is defined and resp.json.error is not defined else '' }}"
    api_ok: >-
      {{ resp.json is defined
         and resp.json.error is not defined
         and resp.json.result.itemids | default([]) | length == 1 }}
    api_erro: >-
      {{ (resp.json.error.message ~ ' - ' ~ (resp.json.error.data | default('')))
         if resp.json is defined and resp.json.error is defined
         else (resp.msg | default('Sem resposta válida da API.')) }}
    status: >-
      {{ linha.status if linha.status != 'PENDENTE'
         else 'SIMULADO' if resp.skipped | default(false)
         else 'CRIADO' if api_ok
         else 'ERRO' }}
    mensagem: >-
      {{ linha.mensagem if linha.status != 'PENDENTE'
         else 'Simulação (aplicar_alteracoes=false): seria criado o item "' ~ linha.payload.name ~ '" com a key ' ~ linha.payload.key_ if status == 'SIMULADO'
         else 'Item criado: ' ~ linha.payload.key_ if status == 'CRIADO'
         else api_erro }}
  loop: "{{ zbx_plano | zip(zbx_create.results) | list }}"
  loop_control:
    label: "{{ item.0.host }} ({{ item.0.tipo }})"
  register: zbx_resultado_loop

- name: Guardar o resultado para o relatório
  ansible.builtin.set_fact:
    resultado_criacao: "{{ zbx_resultado_loop.results | map(attribute='msg') | list }}"

# ============================================================
# 4. Extrair certificados da descrição (ADF)
#
# 4a. Esquema "tipo + data + hosts em lista com marcadores": extração
#     ADF preservada. Vale para chamados de atualização e também para
#     criação, caso o corpo siga o mesmo esquema. Cada certificado

- name: Extrair hosts e validades das descrições do Jira

      {% endfor %}
      {% endif %}

# ------------------------------------------------------------
# 4b. Chamados de CRIAÇÃO. O corpo vem assim:
#
#   Nome completo do certificado: clarocustodioa.claro.com.br
#   Tipo de certificado: Interno
#   Data de renovação: 06/10/2027
#
# Um chamado pode ter VÁRIOS blocos desse tipo (um certificado por
# bloco). Cada linha "Nome ...:" abre um novo certificado; o "Tipo de
# certificado" e a "Data de renovação" seguintes pertencem a ele. O bloco
# fecha quando começa o próximo "Nome" ou a descrição termina.
#
# As linhas podem chegar como parágrafos separados (Formato A) ou como um
# único parágrafo com quebras de linha (hardBreak no ADF, Formato B, que
# vem precedido de um parágrafo "Certificado Novo" e com o rótulo
# "Nome dos certificados:"). Os dois são tratados. O rótulo de host é
# qualquer um que comece com "Nome"; os demais rótulos são reconhecidos
# pelo começo (não dependem de acento). "*." no início do host vira
# "star.", igual à atualização. Bloco sem host ou sem data válida não gera
# certificado (e o chamado sem nenhum certificado é avisado na task seguinte).
# ------------------------------------------------------------
- name: Extrair certificado dos chamados de criação
  ansible.builtin.set_fact:
    certificados_criacao_jira: >-
      {%- set resultado = namespace(certificados=[]) -%}
      {%- for ticket in tickets_criacao | default([]) -%}
      {%- set cert = namespace(host='', tipo='', validade='') -%}
      {%- set pars = namespace(lista=[]) -%}
      {%- for bloco in ticket.fields.description.content | default([]) -%}
      {%- if bloco.type == 'paragraph' -%}
      {%- set pars.lista = pars.lista + [bloco] -%}
      {%- elif bloco.type in ['bulletList', 'orderedList'] -%}
      {%- for li in bloco.content | default([]) -%}
      {%- for pg in li.content | default([]) if pg.type == 'paragraph' -%}
      {%- set pars.lista = pars.lista + [pg] -%}
      {%- endfor -%}
      {%- endfor -%}
      {%- endif -%}
      {%- endfor -%}
      {%- set pares = namespace(lista=[]) -%}
      {%- for bloco in pars.lista -%}
      {%- set t = namespace(texto='') -%}
      {%- for no in bloco.content | default([]) -%}
      {%- if no.type == 'text' -%}
      {%- set t.texto = t.texto ~ no.text -%}
      {%- elif no.type == 'hardBreak' -%}
      {%- set t.texto = t.texto ~ '\n' -%}
      {%- endif -%}
      {%- endfor -%}
      {%- for linha in t.texto.split('\n') if ':' in linha -%}
      {%- set pares.lista = pares.lista + [{'rotulo': linha.split(':', 1)[0] | trim | lower, 'valor': linha.split(':', 1)[1] | trim}] -%}
      {%- endfor -%}
      {%- endfor -%}
      {#- par final "nome" vazio: força o fechamento do último bloco -#}
      {%- set pares.lista = pares.lista + [{'rotulo': 'nome', 'valor': ''}] -%}
      {%- for par in pares.lista -%}
      {%- if par.rotulo.startswith('nome') -%}
      {%- if cert.host -%}
      {%- if cert.validade -%}
      {%- set resultado.certificados = resultado.certificados + [{'ticket': ticket.key, 'operacao': 'CRIACAO', 'host': cert.host | regex_replace('^\\*\\.', 'star.'), 'tipo': cert.tipo, 'validade': cert.validade}] -%}
      {%- endif -%}
      {%- set cert.tipo = '' -%}
      {%- set cert.validade = '' -%}
      {%- endif -%}
      {%- set cert.host = par.valor -%}
      {%- elif par.rotulo.startswith('tipo de certificado') -%}
      {%- set cert.tipo = 'Externo' if par.valor | upper == 'EXTERNO' else 'Interno' if par.valor | upper in ['INTERNO', 'INTERNOS'] else '' -%}
      {%- elif par.rotulo.startswith('data de renova') and par.valor is match('^[0-9]{2}/[0-9]{2}/[0-9]{4}$') -%}
      {%- set cert.validade = par.valor -%}
      {%- endif -%}
      {%- endfor -%}
      {%- endfor -%}
      {{ resultado.certificados }}

- name: Avisar sobre chamados de criação sem certificado reconhecido
  ansible.builtin.debug:
    msg: "Chamado {{ item.key }} ignorado: não foi possível ler host e data de renovação válidos da descrição."
  loop: >-

    label: "{{ item.key }}"
