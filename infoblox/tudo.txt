OBSERVAÇÃO:

A validação da porta 53 depende das ferramentas disponíveis no servidor, podendo utilizar dig, nc (netcat), telnet, nmap ou bash /dev/tcp.
Em servidores que não possuam nenhuma dessas ferramentas, a validação poderá retornar falha sem necessariamente indicar bloqueio de firewall.

Importante: o serviço DNS normalmente utiliza UDP/53 para as consultas. Porém, quando a validação é realizada através do bash (/dev/tcp), o teste passa a depender da comunicação TCP/53.

          Diferentemente do TCP, o UDP não possui confirmação de conexão, portanto o bash não consegue validar de forma confiável a comunicação utilizando /dev/udp. Por esse motivo, quando o servidor não possui ferramentas como dig, nc, telnet ou nmap, o teste pode acabar dependendo exclusivamente da conectividade TCP/53.

          Nestes casos, para que a validação seja concluída com sucesso, pode ser necessário que a comunicação TCP/53 também esteja liberada entre o servidor e os DNS Infoblox, mesmo que a aplicação do DNS utilize predominantemente UDP/53.

- name: Testando conexão na porta 53 com o IP 10.54.44.12
  ansible.builtin.raw: |
    for ip in 10.54.44.12; do
      echo "🔍 Testando DNS $ip:"

      # Teste com nc
      if command -v nc >/dev/null 2>&1 && nc -h 2>&1 | grep -q '\-z'; then
        echo -n "  [nc] "
        nc -z -w2 $ip 53 && echo "OK" || echo "FALHA"

      # Fallback com telnet
      elif command -v telnet >/dev/null 2>&1; then
        echo -n "  [telnet] "
        telnet_output=`echo quit | telnet $ip 53 2>/dev/null`
        echo "$telnet_output" | grep "Escape character" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nc/telnet] Nenhum dos dois disponível"
      fi

      # Teste com dig
      if command -v dig >/dev/null 2>&1; then
        echo -n "  [dig] "
        dig_output=`dig @$ip google.com +time=2 +tries=1`
        echo "$dig_output" | grep "ANSWER SECTION" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [dig] dig não instalado"
      fi

      # Teste com nmap
      if command -v nmap >/dev/null 2>&1; then
        echo -n "  [nmap] "
        nmap_output=`nmap -p 53 -Pn --host-timeout 5s $ip`
        echo "$nmap_output" | grep "53/tcp" | grep "open" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nmap] nmap não instalado"
      fi

    done
  register: dns_one

- name: Testando conexão na porta 53 com o IP 10.54.44.12
  ansible.builtin.raw: |
    for ip in 10.54.44.12; do
      echo "🔍 Testando DNS $ip:"

      # Teste com nc
      if command -v nc >/dev/null 2>&1 && nc -h 2>&1 | grep -q '\-z'; then
        echo -n "  [nc] "
        nc -z -w2 $ip 53 && echo "OK" || echo "FALHA"

      # Fallback com telnet
      elif command -v telnet >/dev/null 2>&1; then
        echo -n "  [telnet] "
        telnet_output=`echo quit | telnet $ip 53 2>/dev/null`
        echo "$telnet_output" | grep "Escape character" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nc/telnet] Nenhum dos dois disponível"
      fi

      # Teste com dig
      if command -v dig >/dev/null 2>&1; then
        echo -n "  [dig] "
        dig_output=`dig @$ip google.com +time=2 +tries=1`
        echo "$dig_output" | grep "ANSWER SECTION" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [dig] dig não instalado"
      fi

      # Teste com nmap
      if command -v nmap >/dev/null 2>&1; then
        echo -n "  [nmap] "
        nmap_output=`nmap -p 53 -Pn --host-timeout 5s $ip`
        echo "$nmap_output" | grep "53/tcp" | grep "open" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nmap] nmap não instalado"
      fi

    done
  register: dns_one

        echo -n "  [nc] "
        nc -z -w2 $ip 53 && echo "OK" || echo "FALHA"

      # Fallback com telnet
      elif command -v telnet >/dev/null 2>&1; then
        echo -n "  [telnet] "
        telnet_output=`echo quit | telnet $ip 53 2>/dev/null`
        echo "$telnet_output" | grep "Escape character" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nc/telnet] Nenhum dos dois disponível"
      fi

      # Teste com dig
      if command -v dig >/dev/null 2>&1; then
        echo -n "  [dig] "
        dig_output=`dig @$ip google.com +time=2 +tries=1`
        echo "$dig_output" | grep "ANSWER SECTION" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [dig] dig não instalado"
      fi

      # Teste com nmap
      if command -v nmap >/dev/null 2>&1; then
        echo -n "  [nmap] "
        nmap_output=`nmap -p 53 -Pn --host-timeout 5s $ip`
        echo "$nmap_output" | grep "53/tcp" | grep "open" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nmap] nmap não instalado"
      fi

    done
  register: dns_two
