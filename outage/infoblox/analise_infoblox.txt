- name: Testando conexão na porta 53 com o IP 10.54.44.12
  ansible.builtin.raw: |
    for ip in 10.54.44.12; do
      echo "🔍 Testando DNS $ip:"

      # Teste com nc
      if command -v nc >/dev/null 2>&1 && nc -h 2>&1 | grep -q '\-z'; then
        echo -n "  [nc] "
        nc -z -w2 $ip 53 && echo "OK" || echo "FALHA"

      # Fallback com telnet
      elif command -v telnet >/dev/null 2>&1; then
        echo -n "  [telnet] "
        telnet_output=`echo quit | telnet $ip 53 2>/dev/null`
        echo "$telnet_output" | grep "Escape character" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nc/telnet] Nenhum dos dois disponível"
      fi

      # Teste com dig
      if command -v dig >/dev/null 2>&1; then
        echo -n "  [dig] "
        dig_output=`dig @$ip google.com +time=2 +tries=1`
        echo "$dig_output" | grep "ANSWER SECTION" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [dig] dig não instalado"
      fi

      # Teste com nmap
      if command -v nmap >/dev/null 2>&1; then
        echo -n "  [nmap] "
        nmap_output=`nmap -p 53 -Pn --host-timeout 5s $ip`
        echo "$nmap_output" | grep "53/tcp" | grep "open" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nmap] nmap não instalado"
      fi

    done
  register: dns_one

- name: Testando conexão na porta 53 com o IP 10.29.177.81
  ansible.builtin.raw: |
    for ip in 10.29.177.81; do
      echo "🔍 Testando DNS $ip:"

      # Teste com nc
      if command -v nc >/dev/null 2>&1 && nc -h 2>&1 | grep -q '\-z'; then
        echo -n "  [nc] "
        nc -z -w2 $ip 53 && echo "OK" || echo "FALHA"

      # Fallback com telnet
      elif command -v telnet >/dev/null 2>&1; then
        echo -n "  [telnet] "
        telnet_output=$(echo quit | telnet $ip 53 2>/dev/null)
        echo "$telnet_output" | grep "Escape character" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nc/telnet] Nenhum dos dois disponível"
      fi

      # Teste com dig
      if command -v dig >/dev/null 2>&1; then
        echo -n "  [dig] "
        dig_output=$(dig @$ip google.com +time=2 +tries=1)
        echo "$dig_output" | grep "ANSWER SECTION" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [dig] dig não instalado"
      fi

      # Teste com nmap
      if command -v nmap >/dev/null 2>&1; then
        echo -n "  [nmap] "
        nmap_output=$(nmap -p 53 -Pn --host-timeout 5s $ip)
        echo "$nmap_output" | grep "53/tcp" | grep "open" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nmap] nmap não instalado"
      fi

    done
  register: dns_one

- name: Testando conexão na porta 53 com o IP 10.54.44.12
  ansible.builtin.raw: |
    for ip in 10.54.44.12; do
      echo "🔍 Testando DNS $ip:"

      # Teste com nc
      if command -v nc >/dev/null 2>&1 && nc -h 2>&1 | grep -q '\-z'; then
        echo -n "  [nc] "
        nc -z -w2 $ip 53 && echo "OK" || echo "FALHA"

      # Fallback com telnet
      elif command -v telnet >/dev/null 2>&1; then
        echo -n "  [telnet] "
        telnet_output=`echo quit | telnet $ip 53 2>/dev/null`
        echo "$telnet_output" | grep "Escape character" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nc/telnet] Nenhum dos dois disponível"
      fi

      # Teste com dig
      if command -v dig >/dev/null 2>&1; then
        echo -n "  [dig] "
        dig_output=`dig @$ip google.com +time=2 +tries=1`
        echo "$dig_output" | grep "ANSWER SECTION" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [dig] dig não instalado"
      fi

      # Teste com nmap
      if command -v nmap >/dev/null 2>&1; then
        echo -n "  [nmap] "
        nmap_output=`nmap -p 53 -Pn --host-timeout 5s $ip`
        echo "$nmap_output" | grep "53/tcp" | grep "open" >/dev/null && echo "OK" || echo "FALHA"
      else
        echo "  [nmap] nmap não instalado"
      fi

    done
  register: dns_one

- name: Validar conectividade SSH e DNS

  ignore_unreachable: true
  ignore_errors: true
  
  vars:
    # IPs DNS por datacenter
    dns_ips:
      Mackenzie:
        - "10.54.44.13"

        - "10.54.44.12"

    # ETAPA 1: TESTAR CONEXÃO SSH
    # -------------------------------------------------------------------------
    - name: SSH | Testar conexão SSH
      ansible.builtin.raw: echo "SSH_OK"
      register: ssh_test
      changed_when: false

    - name: SSH | Registrar status da conexão
      ansible.builtin.set_fact:
        ssh_ok: "{{ ssh_test.stdout is defined and 'SSH_OK' in ssh_test.stdout }}"
        ssh_erro_msg: "{{ ssh_test.msg | default('') }}"

    # -------------------------------------------------------------------------
    - name: SO | Detectar sistema operacional

      register: so_detect
      when: ssh_ok | default(false)

    # - name: SO | Registrar SO detectado

    #   when: ssh_ok | default(false)

            Ubuntu
          {%- else -%}
            Linux
          {%- endif -%}
      when: ssh_ok | default(false)


    # -------------------------------------------------------------------------
    # ETAPA 3: TESTAR PORTA 53 (apenas se SSH OK)
    # -------------------------------------------------------------------------
    - name: DNS | Testar conectividade DNS - IP 1
      ansible.builtin.raw: |
        IP="{{ dns_ips[datacenter][0] }}"

        NC_OK=0
        TELNET_OK=0
        DIG_OK=0
        NMAP_OK=0

        # Teste com nc
        if command -v nc >/dev/null 2>&1 && nc -h 2>&1 | grep -q '\-z'; then
          nc -z -w2 $IP 53 >/dev/null 2>&1 && NC_OK=1
        fi

        # Fallback com telnet
        if command -v telnet >/dev/null 2>&1; then
          telnet_output=$(echo quit | telnet $IP 53 2>/dev/null)
          echo "$telnet_output" | grep -q "Escape character" && TELNET_OK=1
        fi

        # Teste com dig
        if command -v dig >/dev/null 2>&1; then
          dig @$IP google.com +time=2 +tries=1 2>/dev/null | grep -q "^;; SERVER:" && DIG_OK=1
        fi

        # Teste com nmap
        if command -v nmap >/dev/null 2>&1; then
          nmap -p 53 -Pn --host-timeout 5s $IP 2>/dev/null | grep "53/tcp" | grep -q "open" && NMAP_OK=1
        fi

        if [ "$NC_OK" -eq 1 ] || \
           [ "$TELNET_OK" -eq 1 ] || \
           [ "$DIG_OK" -eq 1 ] || \
           [ "$NMAP_OK" -eq 1 ]; then
          echo "DNS1_OK"
        else
          echo "DNS1_FALHA"
        fi
      register: dns1_test

      changed_when: false
